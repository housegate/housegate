# ClickHouse TCP Conn Interface Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Refactor the monolithic `pkg/proxy/proxy.go` (~1800 LOC) into three focused packages — `chproto` (pure codec, thin wrapper over `ch-go/proto`), `chsession` (replayable session state), and a slim `proxy/` (server + relay + hook chain) — preserving current behavior while opening the door to future per-query upstream pooling with state replay.

**Architecture:** Follows the approved spec at [`docs/superpowers/specs/2026-04-21-clickhouse-tcp-conn-interface-design.md`](../specs/2026-04-21-clickhouse-tcp-conn-interface-design.md). Layer A (codec) exposes `ReadPacket(decodeTypes...)` + `Splice` so Data blocks stay zero-copy. Layer C (session) holds `SessionState` as the single source of truth for per-connection state and supports `RebindUpstream(replay=true)` even though MVP never calls it. Relay drives two goroutines and fires `OnHello`/`OnQuery`/`OnException`/`OnClose` hooks; protocol phases are implicit in relay code flow, not an enum.

**Tech Stack:** Go 1.22+, Bazel 8.5.1 + Bzlmod (canonical build), `github.com/ClickHouse/ch-go v0.71.0` (codec primitives, re-used not rewritten), `sentioxyz/sentio-core/common/log` for logging, existing `pkg/cluster` for replica pool, existing `pkg/proxy/rewriter.go` for gRPC rewriter client.

**Canonical build & test invocations:**
- Build: `bazel build //cmd/proxy:housegate`
- All tests: `bazel test //...`
- One test pattern: `bazel test //pkg/chproto:chproto_test --test_filter='TestX'` (similarly for `chsession_test`, `proxy_test`)
- After editing `.go` or adding deps: `bazel mod tidy && bazel run //:gazelle`
- **Do not** rely on `go test ./...` — it panics in the vendored protobuf init. Always use Bazel.

**Test-first discipline:** Each behavior-change task follows: write failing test → run it → see the intended failure → implement minimum → run test → commit. Pure-refactor tasks (moving code without behavior change) commit the move and re-run the existing suite as verification.

```go
// Package chproto is a thin proxy-oriented wrapper around
// github.com/ClickHouse/ch-go/proto. It adds a streaming tagged-union
// ReadPacket, zero-copy Splice for Data/Progress/etc., addendum (chunked)
// negotiation, and a few ClientCode constants ch-go does not expose.
//
// Design: docs/superpowers/specs/2026-04-21-clickhouse-tcp-conn-interface-design.md
package chproto

import (
	"errors"

	"github.com/ClickHouse/ch-go/proto"
)

// Re-exports: downstream packages import chproto only.

	ClientCode  = proto.ClientCode
	ServerCode  = proto.ServerCode
	ClientHello = proto.ClientHello
	ServerHello = proto.ServerHello
	Query       = proto.Query
	Exception   = proto.Exception
	ClientInfo  = proto.ClientInfo
	Setting     = proto.Setting

// Known ClientCode values from ch-go (aliased for locality).
const (
	ClientHelloCode  = proto.ClientCodeHello
	ClientQueryCode  = proto.ClientCodeQuery
	ClientDataCode   = proto.ClientCodeData
	ClientCancelCode = proto.ClientCodeCancel
	ClientPingCode   = proto.ClientCodePing

	ClientScalar                      ClientCode = 7
	ClientIgnoredPartUUIDs            ClientCode = 8
	ClientReadTaskResponse            ClientCode = 9
	ClientMergeTreeReadTaskResponse   ClientCode = 10
	ClientQueryPlan                   ClientCode = 11
	ClientClusterFunctionReadTaskResp ClientCode = 13
)

// Known ServerCode re-exports.
const (
	ServerHelloCode       = proto.ServerCodeHello
	ServerDataCode        = proto.ServerCodeData
	ServerExceptionCode   = proto.ServerCodeException
	ServerProgressCode    = proto.ServerCodeProgress
	ServerPongCode        = proto.ServerCodePong
	ServerEndOfStreamCode = proto.ServerCodeEndOfStream
	ServerProfileInfoCode = proto.ServerCodeProfileInfo
	ServerTotalsCode      = proto.ServerCodeTotals
	ServerExtremesCode    = proto.ServerCodeExtremes
	ServerLogCode         = proto.ServerCodeLog
	ServerTableColumnsCode = proto.ServerCodeTableColumns
)

// Direction tells the Codec which code-set to expect on reads.
type Direction int

const (
	// DirFromClient: reader expects ClientCode packets (client→proxy side).
	DirFromClient Direction = iota
	// DirToUpstream: reader expects ServerCode packets (upstream→proxy side).
	DirToUpstream
)

// Packet is the tagged union produced by Codec.ReadPacket.
//
// Exactly one of (Decoded, Raw) is populated:
//   - Decoded != nil when Type was in the decodeTypes list passed to ReadPacket
//   - Raw   != nil when Type was NOT requested (caller hands it to Splice)
type Packet struct {
	// Type is a ClientCode or ServerCode value cast to uint64 (ch-go uses byte
	// for these enums; uint64 keeps the API forward-compatible if CH ever
	// extends past 255).
	Type    uint64
	// RawLen is the total on-wire length of the packet including its leading
	// VarUInt type byte.
	RawLen  int
	// Decoded holds a typed struct pointer: *ClientHello, *ServerHello,
	// *Query, *Exception. nil for packets not in the decodeTypes set.
	Decoded any
	// Raw holds the accumulated on-wire bytes of the packet (including the
	// leading type VarUInt) when Decoded is nil. Intended to be handed to
	// Codec.Splice.
	Raw []byte
}

// Sentinel errors.
var (
	ErrMalformed     = errors.New("chproto: malformed packet")
	ErrUnknownPacket = errors.New("chproto: unknown packet type")
	ErrDecode        = errors.New("chproto: decode failed")
)
```

```go
package chproto

import (
	"bufio"
	"io"
	"sync/atomic"

	"github.com/ClickHouse/ch-go/proto"
)

// Codec wraps one side of a ClickHouse TCP connection.
//
// One Codec per net.Conn. Reads are buffered via bufio.Reader (128KB default,
// matching current proxy.go). Writes go directly to the underlying io.Writer
// because the proxy relay does not need write-side buffering beyond what
// ch-go's *proto.Buffer provides.
type Codec struct {
	r    *proto.Reader
	br   *bufio.Reader
	w    io.Writer
	conn io.ReadWriter
	rev  atomic.Int64 // negotiated protocol revision; 0 until SetRevision
	dir  Direction
}

const defaultBufSize = 131072 // 128KB, matches legacy proxy.go

// NewCodec constructs a Codec wrapping conn. direction controls the code-set
// ReadPacket expects.
func NewCodec(conn io.ReadWriter, direction Direction) *Codec {
	br := bufio.NewReaderSize(conn, defaultBufSize)
	return &Codec{
		r:    proto.NewReader(br),
		br:   br,
		w:    conn,
		conn: conn,
		dir:  direction,
	}
}

// Conn returns the underlying io.ReadWriter (for Splice targets and
// Close from the session).
func (c *Codec) Conn() io.ReadWriter { return c.conn }

// SetRevision records the negotiated protocol revision. Called by the caller
// after handshake completes; affects decoding of packets whose layout varies
// with revision (Query primarily).
func (c *Codec) SetRevision(rev int) { c.rev.Store(int64(rev)) }

// Revision returns the last revision set by SetRevision, or 0 pre-handshake.
func (c *Codec) Revision() int { return int(c.rev.Load()) }
```

```go
package chproto

import (
	"fmt"

	"github.com/ClickHouse/ch-go/proto"
)

// decodeClientHello reads a ClientHello body from the Codec's proto.Reader.
// The caller must have already consumed the type VarUInt from the wire.
func (c *Codec) decodeClientHello() (*ClientHello, error) {
	h := &proto.ClientHello{}
	if err := h.Decode(c.r); err != nil {
		return nil, fmt.Errorf("%w: client hello: %v", ErrDecode, err)
	}
	return h, nil
}

// decodeServerHello mirrors decodeClientHello for ServerHello.
// revision=0 is acceptable pre-handshake; ch-go's ServerHello.Decode handles
// the version-sensitive fields internally based on Revision.
func (c *Codec) decodeServerHello() (*ServerHello, error) {
	h := &proto.ServerHello{}
	if err := h.Decode(c.r); err != nil {
		return nil, fmt.Errorf("%w: server hello: %v", ErrDecode, err)
	}
	return h, nil
}

// WriteClientHello encodes a ClientHello onto the wire including the leading
// type VarUInt.
func (c *Codec) WriteClientHello(h *ClientHello) error {
	var buf proto.Buffer
	buf.PutUVarInt(uint64(proto.ClientCodeHello))
	h.Encode(&buf)
	_, err := c.w.Write(buf.Buf)
	return err
}

// WriteServerHello mirrors WriteClientHello for ServerHello.
func (c *Codec) WriteServerHello(h *ServerHello) error {
	var buf proto.Buffer
	buf.PutUVarInt(uint64(proto.ServerCodeHello))
	h.EncodeAware(&buf, c.Revision())
	_, err := c.w.Write(buf.Buf)
	return err
}
```

- [ ] **Step 4: Add `ReadPacket` to `codec.go`.** Buffered-probing strategy: read the type VarUInt; if it's in `decodeTypes`, dispatch to the type-specific decoder; otherwise, accumulate bytes from the packet into a raw buffer.

For this task we implement only the Hello branches plus a "raw passthrough" fallback. Query, Exception, and Data come in later tasks.

Append to `pkg/chproto/codec.go`:

```go
import (
	// … existing imports …
	"fmt"
)

// ReadPacket reads the next packet from the wire.
//
// If the packet's type appears in decodeTypes, it is decoded into a typed
// struct and returned in Packet.Decoded. Otherwise, the entire packet
// (including its leading type VarUInt) is accumulated into Packet.Raw for
// zero-copy Splice.
//
// Known sentinels:
//   io.EOF           — peer closed cleanly
//   ErrMalformed     — protocol error; caller must close
//   ErrUnknownPacket — type not in known table; Packet still carries Raw bytes
//                      so caller can Splice through for forward-compat
//   ErrDecode        — decode of a requested type failed; per spec D8, caller
//                      falls back to Splice(Raw) — this function populates
//                      both Raw and the Decode error.
func (c *Codec) ReadPacket(decodeTypes ...uint64) (*Packet, error) {
	// Capture the byte stream from this point so we can return Raw for
	// non-decoded packets or fail-open on ErrDecode.
	tee := newTeeProtoReader(c.br)
	typeVar, err := proto.NewReader(tee).ReadUVarInt()
	if err != nil {
		return nil, err // includes io.EOF
	}

	wantDecode := false
	for _, t := range decodeTypes {
		if t == typeVar {
			wantDecode = true
			break
		}
	}

	// Swap the codec's reader to the tee'd one so decoder methods see the
	// same stream. We restore afterwards.
	restore := c.r
	c.r = proto.NewReader(tee)
	defer func() { c.r = restore }()

	if !wantDecode {
		// Consume bytes belonging to this packet by dispatching to a
		// length-walker (not decoder). For v1 only Hello/Query/Exception/Data
		// need walkers — Data is added in a later task; the other
		// zero-copy packets (Ping/Cancel/Pong/Progress/EOS/…) have fixed
		// or trivially derivable lengths.
		if err := c.skipPacketBody(typeVar); err != nil {
			return &Packet{Type: typeVar, Raw: tee.captured(), RawLen: len(tee.captured())}, err
		}
		raw := tee.captured()
		return &Packet{Type: typeVar, RawLen: len(raw), Raw: raw}, nil
	}

	var decoded any
	var decErr error
	switch typeVar {
	case uint64(proto.ClientCodeHello):
		decoded, decErr = c.decodeClientHello()
	case uint64(proto.ServerCodeHello):
		decoded, decErr = c.decodeServerHello()
	// Query / Exception / … added in later tasks.
	default:
		decErr = fmt.Errorf("%w: type=%d", ErrUnknownPacket, typeVar)
	}

	raw := tee.captured()
	if decErr != nil {
		// fail-open: return the Raw bytes AND the decode error; caller
		// (Relay) treats ErrDecode as a signal to Splice(Raw) through.
		return &Packet{Type: typeVar, RawLen: len(raw), Raw: raw}, decErr
	}
	return &Packet{Type: typeVar, RawLen: len(raw), Decoded: decoded}, nil
}

// skipPacketBody consumes bytes belonging to a packet whose type is not
// in the decode set. For packet types that have fixed / trivially computed
// lengths (Ping, Cancel, Pong, EndOfStream, Progress) this is a no-op for
// body-less types or reads the well-defined body size.
//
// For Hello/Query/Exception/Data, length isn't fixed — we decode-to-discard
// using ch-go helpers when we can, or Splice-specific framing for Data (added
// in a later task). v1 of this function supports only the fixed-size
// zero-body packets plus Ping/Pong/Cancel; adding the other types is done
// in the task that introduces each.
func (c *Codec) skipPacketBody(typeVar uint64) error {
	switch typeVar {
	case uint64(proto.ClientCodePing),
		uint64(proto.ClientCodeCancel),
		uint64(proto.ServerCodePong),
		uint64(proto.ServerCodeEndOfStream):
		return nil // zero-length bodies
	}
	return fmt.Errorf("%w: skip body for type=%d not yet supported", ErrUnknownPacket, typeVar)
}
```

- [ ] **Step 5: Add the `teeProtoReader` helper.**

Append to `pkg/chproto/codec.go`:

```go
// teeProtoReader wraps an io.Reader, capturing every byte read into an
// internal buffer so the caller can retrieve the exact byte stream after
// a successful (or partially successful) decode. It is the mechanism that
// lets ReadPacket hand the Raw bytes to Splice when a decode was not
// requested or failed.
type teeProtoReader struct {
	src *bufio.Reader
	cap []byte
}

func newTeeProtoReader(src *bufio.Reader) *teeProtoReader {
	return &teeProtoReader{src: src, cap: make([]byte, 0, 1024)}
}

func (t *teeProtoReader) Read(p []byte) (int, error) {
	n, err := t.src.Read(p)
	if n > 0 {
		t.cap = append(t.cap, p[:n]...)
	}
	return n, err
}

func (t *teeProtoReader) captured() []byte { return t.cap }
```

```go
package chproto

import (
	"fmt"

	"github.com/ClickHouse/ch-go/proto"

// decodeQuery reads a Query body at the current revision.
//
// The caller must have already consumed the type VarUInt. The negotiated
// revision governs which fields are present (ClientInfo layout varies
// by revision in ch-go).
func (c *Codec) decodeQuery() (*Query, error) {
	rev := c.Revision()
	if rev == 0 {
		return nil, fmt.Errorf("%w: Query decode requires SetRevision first", ErrMalformed)
	}
	q := &proto.Query{}
	if err := q.DecodeAware(c.r, rev); err != nil {
		return nil, fmt.Errorf("%w: query: %v", ErrDecode, err)
	}
	return q, nil
}

// WriteQuery encodes a Query onto the wire including the leading type VarUInt,
// using the codec's current revision.
func (c *Codec) WriteQuery(q *Query) error {
	rev := c.Revision()
	if rev == 0 {
		return fmt.Errorf("%w: WriteQuery requires SetRevision first", ErrMalformed)
	}
	var buf proto.Buffer
	buf.PutUVarInt(uint64(proto.ClientCodeQuery))
	q.EncodeAware(&buf, rev)
	_, err := c.w.Write(buf.Buf)
	return err
}
```

```go
// Splice forwards a Packet read by ReadPacket to dst, byte-for-byte.
//
// Two paths:
//   - Decoded != nil: caller used a modifying encoder already; Splice is a
//     no-op error. This is an API-usage violation because the caller should
//     have called WriteX(Decoded) instead.
//   - Raw != nil: write Raw directly. For Data packets, Raw already contains
//     the framed bytes (ReadPacket's raw-capture path walked frames via
//     skipPacketBody → walkDataBlock).
func (c *Codec) Splice(dst io.Writer, p *Packet) error {
	if p == nil {
		return fmt.Errorf("%w: nil packet", ErrMalformed)
	}
	if p.Decoded != nil {
		return fmt.Errorf("%w: Splice called on decoded packet; use a typed Write* instead", ErrMalformed)
	}
	_, err := dst.Write(p.Raw)
	return err
}
```

```go
func (c *Codec) skipPacketBody(typeVar uint64) error {
```

```go
// AddendumOpts is the set of chunked-mode preferences the caller proposes
// to the peer during addendum negotiation.
type AddendumOpts struct {
	ProposedRecv string // "notchunked" | "chunked" | "chunked_optional"
	ProposedSend string
}

// AddendumResult is the outcome of addendum negotiation with the peer.
type AddendumResult struct {
	NegotiatedRecv string
	NegotiatedSend string
}

// NegotiateAddendum performs the chunked-protocol addendum handshake on
// the CLIENT-facing side: reads the client's proposal, resolves it against
// our AddendumOpts using the same rules as legacy resolveChunkedMode, and
// writes our response. Used by Relay immediately after the ServerHello
// exchange if SupportsAddendum(rev).
func (c *Codec) NegotiateAddendum(opts AddendumOpts) (AddendumResult, error) {
	// The addendum bytes layout is identical to the one handled today in
	// pkg/proxy/proxy.go's handshake path (search "addendum" in that file):
	// the client sends two length-prefixed strings (recv_mode, send_mode),
	// we reply with the same pair of strings after resolution.
	clientRecv, err := c.r.Str()
	if err != nil {
		return AddendumResult{}, fmt.Errorf("addendum: read client recv: %w", err)
	}
	clientSend, err := c.r.Str()
	if err != nil {
		return AddendumResult{}, fmt.Errorf("addendum: read client send: %w", err)
	}

	res := AddendumResult{
		NegotiatedRecv: resolveChunkedMode(opts.ProposedRecv, clientSend),
		NegotiatedSend: resolveChunkedMode(opts.ProposedSend, clientRecv),
	}
	return res, c.SendAddendum(res)
}

// SendAddendum writes the resolved addendum result to the peer (used on the
// upstream side after we read the ServerHello's addendum, or on the client
// side at the end of NegotiateAddendum).
func (c *Codec) SendAddendum(res AddendumResult) error {
	var buf proto.Buffer
	buf.PutString(res.NegotiatedRecv)
	buf.PutString(res.NegotiatedSend)
	_, err := c.w.Write(buf.Buf)
	return err
}
```

```go
	ClientHostname    string
	ClientVersion     VersionTriple
	ClientRevision    int
	ServerDisplayName string
	Timezone          string
	ChunkedRecv       string
	ChunkedSend       string

	AuthenticatedUser string
	MappedUser        string
	MappedPassword    string
	Identity          IdentityClaims

	Database string
	Settings map[string]chproto.Setting

	LastQueryID      string
	LastClientInfo   *chproto.ClientInfo
	LastRewriteArgs  any
	HasActiveRewrite bool
}

// Snapshot returns a copy of the state under a read lock. The Settings map
// is shallow-copied so the caller cannot mutate the original.
func (s *SessionState) Snapshot() SessionStateSnapshot {
	s.mu.RLock()
	defer s.mu.RUnlock()
	snap := SessionStateSnapshot{
		ClientHostname:    s.ClientHostname,
		ClientVersion:     s.ClientVersion,
		ClientRevision:    s.ClientRevision,
		ServerDisplayName: s.ServerDisplayName,
		Timezone:          s.Timezone,
		ChunkedRecv:       s.ChunkedRecv,
		ChunkedSend:       s.ChunkedSend,
		AuthenticatedUser: s.AuthenticatedUser,
		MappedUser:        s.MappedUser,
		MappedPassword:    s.MappedPassword,
		Identity:          s.Identity,
		Database:          s.Database,
		LastQueryID:       s.LastQueryID,
		LastClientInfo:    s.LastClientInfo,
		LastRewriteArgs:   s.LastRewriteArgs,
		HasActiveRewrite:  s.HasActiveRewrite,
	}
	if len(s.Settings) > 0 {
		snap.Settings = make(map[string]chproto.Setting, len(s.Settings))
		for k, v := range s.Settings {
			snap.Settings[k] = v
		}
	}
	return snap
}

// SetDatabase updates the current database. Called by StateTrackerPlugin
// when it observes a USE statement.
func (s *SessionState) SetDatabase(db string) {
	s.mu.Lock()
	s.Database = db
	s.mu.Unlock()
}

// AddSetting records a session-level SETTING. Called by StateTrackerPlugin
// when it observes a SET statement. Per spec D7, Query.Settings (per-query
// payload) must NOT flow through here.
func (s *SessionState) AddSetting(k string, v chproto.Setting) {
	s.mu.Lock()
	if s.Settings == nil {
		s.Settings = make(map[string]chproto.Setting)
	}
	s.Settings[k] = v
	s.mu.Unlock()
}

// MarkActiveRewrite flags the current in-flight query as having undergone
// rewriting. Relay uses the flag to decide whether to request Exception
// decoding on the upstream-to-client path (so reverse-mapping can run).
func (s *SessionState) MarkActiveRewrite(args any) {
	s.mu.Lock()
	s.HasActiveRewrite = true
	s.LastRewriteArgs = args
	s.mu.Unlock()
}

// ClearActiveRewrite is called when the query completes (EndOfStream or
// Exception).
func (s *SessionState) ClearActiveRewrite() {
	s.mu.Lock()
	s.HasActiveRewrite = false
	s.mu.Unlock()
}
```

```go
package chsession

import (
	"context"
	"fmt"
	"net"
	"sync"
	"sync/atomic"

// Session carries per-connection state and the two codecs (client + upstream).
// It is passive — it does not run the packet loop. Relay does.
//
// MVP: BindUpstream is called once at session start; RebindUpstream is
// never called in production code paths but is implemented and tested so
// future C3 (per-query pool borrow with state replay) can adopt it without
// interface churn.
type Session interface {
	ID() int64
	State() *SessionState
	Client() *chproto.Codec
	Upstream() *chproto.Codec
	RemoteAddr() net.Addr
	Close() error

	BindUpstream(ctx context.Context, up *chproto.Codec) error
	RebindUpstream(ctx context.Context, newUp *chproto.Codec, replayState bool) error
}

type sessionImpl struct {
	id         int64
	state      *SessionState
	client     *chproto.Codec
	up         atomic.Pointer[chproto.Codec]
	clientConn net.Conn
	closeOnce  sync.Once
	closeErr   error
}

// New wraps clientConn in a Codec and constructs a Session. No network
// traffic occurs; handshake is Relay's responsibility.
func New(id int64, clientConn net.Conn) Session {
	return &sessionImpl{
		id:         id,
		state:      NewSessionState(),
		client:     chproto.NewCodec(clientConn, chproto.DirFromClient),
		clientConn: clientConn,
	}
}

func (s *sessionImpl) ID() int64            { return s.id }
func (s *sessionImpl) State() *SessionState { return s.state }
func (s *sessionImpl) Client() *chproto.Codec { return s.client }
func (s *sessionImpl) RemoteAddr() net.Addr {
	if s.clientConn == nil {
		return nil
	}
	return s.clientConn.RemoteAddr()
}

func (s *sessionImpl) Upstream() *chproto.Codec {
	return s.up.Load()
}

// BindUpstream sets the upstream codec. Errors if already bound.
func (s *sessionImpl) BindUpstream(_ context.Context, up *chproto.Codec) error {
	if !s.up.CompareAndSwap(nil, up) {
		return fmt.Errorf("%w: upstream already bound", ErrRebindDenied)
	}
	return nil
}

// RebindUpstream atomically swaps the upstream codec. The caller is expected
// to have already performed the handshake against newUp; when replayState is
// true, SessionState.Replay is invoked after the swap.
//
// MVP has no production caller. See spec §5.4 for semantics.
func (s *sessionImpl) RebindUpstream(ctx context.Context, newUp *chproto.Codec, replayState bool) error {
	if newUp == nil {
		return fmt.Errorf("%w: nil upstream", ErrRebindDenied)
	}
	old := s.up.Swap(newUp)
	if old != nil {
		if closer, ok := old.Conn().(interface{ Close() error }); ok {
			go func() { _ = closer.Close() }()
		}
	}
	if replayState {
		return s.state.Replay(ctx, newUp)
	}
	return nil
}

// Close tears down the session. Idempotent.
func (s *sessionImpl) Close() error {
	s.closeOnce.Do(func() {
		if s.clientConn != nil {
			_ = s.clientConn.Close()
		}
		if up := s.up.Load(); up != nil {
			if closer, ok := up.Conn().(interface{ Close() error }); ok {
				_ = closer.Close()
			}
		}
	})
	return s.closeErr
}
```

```go
// Replay re-applies the replayable subset (Database, Settings) to a newly
// bound upstream by sending USE and SET queries through it.
//
// The queries use the upstream codec's current revision; the caller (Session.
// RebindUpstream) is responsible for completing the handshake before calling

// and testable.
func (s *SessionState) Replay(ctx context.Context, up *chproto.Codec) error {
	s.mu.RLock()
	db := s.Database
	settings := make(map[string]chproto.Setting, len(s.Settings))
	for k, v := range s.Settings {
		settings[k] = v
	}

		if err := up.WriteQuery(q); err != nil {
			return fmt.Errorf("replay USE: %w", err)
		}
	}

	keys := make([]string, 0, len(settings))
	for k := range settings {
		keys = append(keys, k)
	}
	sort.Strings(keys)
	for _, k := range keys {
		v := settings[k]
		q := &chproto.Query{Body: fmt.Sprintf("SET %s=%v", k, v.Value)}
		if err := up.WriteQuery(q); err != nil {
			return fmt.Errorf("replay SET %s: %w", k, err)
		}
	}
	return nil
}
```

```go
package proxy

import (
	"context"
	"fmt"

	log "sentioxyz/sentio-core/common/log"

	"housegate/housegate/pkg/chproto"
	"housegate/housegate/pkg/chsession"
)

// Relay drives the bidirectional packet loop between a client and its
// bound upstream. It is constructed once per session and never shared.
//
// Design: docs/superpowers/specs/2026-04-21-clickhouse-tcp-conn-interface-design.md
type Relay struct {
	sess  chsession.Session
	hooks Hooks
}

// NewRelay constructs a Relay. The session must already have an upstream
// bound via sess.BindUpstream.
func NewRelay(sess chsession.Session, hooks Hooks) *Relay {
	return &Relay{sess: sess, hooks: hooks}
}

// handshake performs the three-phase ClickHouse handshake:
//   1. Read the client's ClientHello; run OnHello plugins (which may inject
//      credentials, strip a __route__ prefix, etc.).
//   2. Forward the possibly-mutated hello to the upstream; read its
//      ServerHello; echo the ServerHello back to the client.
//   3. If the negotiated revision supports it, run addendum negotiation on
//      both sides.
//
// On completion, Codec revisions are set and SessionState is populated.
func (r *Relay) handshake(ctx context.Context) error {
	client := r.sess.Client()
	upstream := r.sess.Upstream()
	if upstream == nil {
		return chsession.ErrNoUpstream
	}

	pkt, err := client.ReadPacket(uint64(chproto.ClientHelloCode))
	if err != nil {
		return fmt.Errorf("read client hello: %w", err)
	}
	hello, ok := pkt.Decoded.(*chproto.ClientHello)
	if !ok {
		// Per D8, fail-open. For Hello specifically, without a valid decode
		// we cannot proceed — close the connection.
		return fmt.Errorf("client hello: %w", chproto.ErrDecode)
	}

	if err := r.hooks.OnHello(ctx, r.sess, hello); err != nil {
		return fmt.Errorf("on-hello: %w", err)
	}

	// Record pre-negotiation fields; revision is finalized below.
	state := r.sess.State()
	state.ClientHostname = hello.Name
	state.ClientVersion = chsession.VersionTriple{
		Major: int(hello.Major),
		Minor: int(hello.Minor),
		Patch: 0,
	}
	state.Database = hello.Database
	if state.MappedUser == "" { // plugins may have set it
		state.MappedUser = hello.User
	}
	if state.MappedPassword == "" {
		state.MappedPassword = hello.Password
	}

	if err := upstream.WriteClientHello(hello); err != nil {
		return fmt.Errorf("forward client hello: %w", err)
	}

	srvPkt, err := upstream.ReadPacket(uint64(chproto.ServerHelloCode))
	if err != nil {
		return fmt.Errorf("read server hello: %w", err)
	}
	srv, ok := srvPkt.Decoded.(*chproto.ServerHello)
	if !ok {
		return fmt.Errorf("server hello: %w", chproto.ErrDecode)
	}

	rev := int(hello.ProtocolVersion)
	if int(srv.Revision) < rev {
		rev = int(srv.Revision)
	}
	client.SetRevision(rev)
	upstream.SetRevision(rev)
	state.ClientRevision = rev
	state.ServerDisplayName = srv.DisplayName
	state.Timezone = srv.Timezone

	if err := client.WriteServerHello(srv); err != nil {
		return fmt.Errorf("echo server hello: %w", err)
	}

	if chproto.SupportsAddendum(rev) {
		res, err := client.NegotiateAddendum(chproto.AddendumOpts{
			ProposedRecv: "chunked_optional",
			ProposedSend: "chunked_optional",
		})
		if err != nil {
			return fmt.Errorf("client addendum: %w", err)
		}
		state.ChunkedRecv = res.NegotiatedRecv
		state.ChunkedSend = res.NegotiatedSend
		if err := upstream.SendAddendum(res); err != nil {
			return fmt.Errorf("upstream addendum: %w", err)
		}
	}

	log.Infow("chsession handshake complete",
		"id", r.sess.ID(),
		"user", state.MappedUser,
		"database", state.Database,
		"revision", rev,
	)
	return nil
}
```

```go
import (
	// … existing imports …
	"errors"
	"io"
)

// Run performs the handshake, then drives two goroutines: client→upstream
// and upstream→client. It returns when either side terminates. Exactly one
// Close is performed on the session, via Run's defer.
func (r *Relay) Run(ctx context.Context) error {
	if err := r.handshake(ctx); err != nil {
		_ = r.sess.Close()
		r.hooks.OnClose(r.sess)
		return err
	}
	defer func() {
		_ = r.sess.Close()
		r.hooks.OnClose(r.sess)
	}()

	errCh := make(chan error, 2)
	go func() { errCh <- r.clientToUpstream(ctx) }()
	go func() { errCh <- r.upstreamToClient(ctx) }()

	first := <-errCh
	_ = r.sess.Close() // nudge the second goroutine to exit
	<-errCh
	if errors.Is(first, io.EOF) || first == nil {
		return nil
	}
	return first
}

// clientToUpstream reads client packets, decoding Query for the OnQuery
// plugin chain; everything else is spliced through unchanged.
func (r *Relay) clientToUpstream(ctx context.Context) error {
	client := r.sess.Client()
	for {
		up := r.sess.Upstream() // atomic: picks up rebinds (future C3)
		if up == nil {
			return chsession.ErrNoUpstream
		}

		pkt, decErr := client.ReadPacket(uint64(chproto.ClientQueryCode))
		if errors.Is(decErr, io.EOF) {
			return io.EOF
		}
		if pkt == nil {
			return decErr
		}
		if decErr != nil && !errors.Is(decErr, chproto.ErrDecode) {
			return decErr
		}

		if decErr == nil && pkt.Decoded != nil && pkt.Type == uint64(chproto.ClientQueryCode) {
			q := pkt.Decoded.(*chproto.Query)
			qctx := &QueryContext{
				Session:     r.sess,
				OriginalSQL: q.Body,
				Query:       q,
				Values:      make(map[string]any),
			}
			if err := r.hooks.OnQuery(ctx, qctx); err != nil {
				r.writeExceptionToClient(err)
				continue
			}
			if err := up.WriteQuery(qctx.Query); err != nil {
				return fmt.Errorf("forward query: %w", err)
			}
			continue
		}

		// Splice any non-decoded / decode-failed packet.
		if err := client.Splice(up.Conn(), pkt); err != nil {
			return fmt.Errorf("splice client→upstream type=%d: %w", pkt.Type, err)
		}
	}
}

// upstreamToClient reads upstream packets. Exception is decoded only when
// SessionState.HasActiveRewrite so OnException can reverse-map the message.
// On EndOfStream, HasActiveRewrite is cleared.
func (r *Relay) upstreamToClient(ctx context.Context) error {
	client := r.sess.Client()
	for {
		up := r.sess.Upstream()
		if up == nil {
			return chsession.ErrNoUpstream
		}

		var decodeTypes []uint64
		if r.sess.State().Snapshot().HasActiveRewrite {
			decodeTypes = append(decodeTypes, uint64(chproto.ServerExceptionCode))
		}

		pkt, decErr := up.ReadPacket(decodeTypes...)
		if errors.Is(decErr, io.EOF) {
			return io.EOF
		}
		if pkt == nil {
			return decErr
		}
		if decErr != nil && !errors.Is(decErr, chproto.ErrDecode) {
			return decErr
		}

		if pkt.Type == uint64(chproto.ServerEndOfStreamCode) {
			r.sess.State().ClearActiveRewrite()
		}

		if decErr == nil && pkt.Decoded != nil && pkt.Type == uint64(chproto.ServerExceptionCode) {
			exc := pkt.Decoded.(*chproto.Exception)
			_ = r.hooks.OnException(ctx, r.sess, exc)
			if err := client.WriteException(exc); err != nil {
				return fmt.Errorf("forward exception: %w", err)
			}
			r.sess.State().ClearActiveRewrite()
			continue
		}

		if err := up.Splice(client.Conn(), pkt); err != nil {
			return fmt.Errorf("splice upstream→client type=%d: %w", pkt.Type, err)
		}
	}
}

// writeExceptionToClient converts a plugin error into a synthetic
// ClickHouse Exception and writes it to the client side. Used when the
// OnQuery chain short-circuits (e.g. JWS invalid, rate-limit).
func (r *Relay) writeExceptionToClient(pluginErr error) {
	exc := &chproto.Exception{
		Code:    403, // ClickHouse AUTHENTICATION_FAILED; generic plugin-reject
		Name:    "DB::Exception",
		Message: pluginErr.Error(),
	}
	if err := r.sess.Client().WriteException(exc); err != nil {
		log.Warnw("failed to write plugin-error exception", "err", err)
	}
}
```

```go
package proxy

import (
	"context"
	"regexp"

	"housegate/housegate/pkg/chproto"
)

// StateTrackerPlugin detects USE <database> and SET key=value statements
// and mirrors them into SessionState.
//
// We intentionally use regex, NOT the AST-aware rewriter gRPC service:
//   - USE and SET syntax is trivially stable (covered by ClickHouse
//     regression tests for many years).
//   - The rewriter gRPC round-trip is ~1-5ms per call; avoiding it on
//     every query keeps the hot path cheap.
//   - Any false positives (e.g. the literal string "USE" inside a comment
//     or a quoted string) are harmless because the same SQL still goes to
//     the real ClickHouse, whose AST parse is authoritative; we only
//     mirror state for our own rewrite-decision purposes.
//
// Limitations documented for operators:
//   - Only a single SET per query is recognized (compound
//     "SET a=1, b=2" would need parsing); users who need compound SETs
//     should issue one statement per setting.
//   - USE must be a top-level statement, not nested (rare but possible
//     in CH multi-statement submissions).
type StateTrackerPlugin struct{}

var (
	reUse = regexp.MustCompile(`(?i)^\s*USE\s+([a-zA-Z_][a-zA-Z0-9_]*)\s*;?\s*$`)
	reSet = regexp.MustCompile(`(?i)^\s*SET\s+([a-zA-Z_]\w*)\s*=\s*(.+?)\s*;?\s*$`)
)

func (p *StateTrackerPlugin) OnQuery(_ context.Context, qctx *QueryContext) error {
	sql := qctx.OriginalSQL
	if qctx.RewrittenSQL != "" {
		sql = qctx.RewrittenSQL
	}
	if m := reUse.FindStringSubmatch(sql); m != nil {
		qctx.Session.State().SetDatabase(m[1])
		return nil
	}
	if m := reSet.FindStringSubmatch(sql); m != nil {
		qctx.Session.State().AddSetting(m[1], chproto.Setting{
			Key:   m[1],
			Value: m[2],
		})
	}
	return nil
}
```

```go
// QueryPlugin participates in the OnQuery chain. Plugins are invoked in
// slice order; the first error short-circuits and is returned as the
// hook's error (Relay converts it to a synthetic Exception toward the
// client).
type QueryPlugin interface {
	OnQuery(ctx context.Context, qctx *QueryContext) error
}

// HelloPlugin participates in the OnHello chain.
type HelloPlugin interface {
	OnHello(ctx context.Context, sess chsession.Session, hello *chproto.ClientHello) error
}

// ExceptionPlugin participates in the OnException chain.
type ExceptionPlugin interface {
	OnException(ctx context.Context, sess chsession.Session, exc *chproto.Exception) error
}

// ClosePlugin participates in OnClose.
type ClosePlugin interface {
	OnClose(sess chsession.Session)
}

// PluginChain composes lists of plugins into a Hooks implementation.
//
// Ordering rule: the first non-nil error in OnHello / OnQuery short-circuits.
// OnException and OnClose run every plugin regardless of errors.
type PluginChain struct {
	HelloPlugins     []HelloPlugin
	QueryPlugins     []QueryPlugin
	ExceptionPlugins []ExceptionPlugin
	ClosePlugins     []ClosePlugin
}

func (c *PluginChain) OnHello(ctx context.Context, sess chsession.Session, h *chproto.ClientHello) error {
	for _, p := range c.HelloPlugins {
		if err := p.OnHello(ctx, sess, h); err != nil {
			return err
		}
	}
	return nil
}

func (c *PluginChain) OnQuery(ctx context.Context, qctx *QueryContext) error {
	for _, p := range c.QueryPlugins {
		if err := p.OnQuery(ctx, qctx); err != nil {
			return err
		}
	}
	return nil
}

func (c *PluginChain) OnException(ctx context.Context, sess chsession.Session, exc *chproto.Exception) error {
	var first error
	for _, p := range c.ExceptionPlugins {
		if err := p.OnException(ctx, sess, exc); err != nil && first == nil {
			first = err
		}
	}
	return first
}

func (c *PluginChain) OnClose(sess chsession.Session) {
	for _, p := range c.ClosePlugins {
		p.OnClose(sess)
	}
}

var _ Hooks = (*PluginChain)(nil)
```

```go
type CredentialInjectorPlugin struct {

	if p.Validator == nil {
		return nil

	claims, err := p.Validator.ValidateQuery(ctx, qctx.Session, qctx.Query)
	if err != nil {
		return fmt.Errorf("jws-validator: %w", err)

		TenantID: claims.TenantID,
		UserID:   claims.UserID,

// When rewriting happens, the plugin marks the session HasActiveRewrite
// so Relay will decode upstream Exception packets for reverse-mapping.
type RewriterPlugin struct {
	Rewriter Rewriter // existing interface in pkg/proxy/rewriter.go
}

func (p *RewriterPlugin) OnQuery(ctx context.Context, qctx *QueryContext) error {
	if p.Rewriter == nil {
		return nil

	out, err := p.Rewriter.Rewrite(ctx, qctx.Query.Body)
	if err != nil {
		// Fail-open per spec §6.1: forward original SQL, emit warn via logger.
		log.Warnw("rewrite failed; forwarding original SQL", "err", err, "sql", qctx.Query.Body)
		return nil

		qctx.Query.Body = out.SQL
		qctx.RewrittenSQL = out.SQL
		qctx.RewriteArgs = out.Args
		qctx.Session.State().MarkActiveRewrite(out.Args)
	}
	return nil
}

}

		return nil

	args, _ := sess.State().Snapshot().LastRewriteArgs.(RewriteArgs) // existing type
```

```go
package proxy

import (
	"context"
	"net"
	"sync/atomic"

	log "sentioxyz/sentio-core/common/log"

	"housegate/housegate/pkg/chproto"
	"housegate/housegate/pkg/chsession"
	"housegate/housegate/pkg/cluster"
)

// Server accepts TCP connections, constructs a Session per connection, and
// hands it to Relay.
//
// UpstreamDialer is mode-specific (server / sidecar / forwarding-only) and
// supplied at construction time. Hooks is the composed PluginChain.
type Server struct {
	Hooks          Hooks
	UpstreamDialer func(ctx context.Context, sess chsession.Session) (*chproto.Codec, error)

	nextID atomic.Int64
	log    *log.Logger // sentio-core zap wrapper
}

func NewServer(hooks Hooks, dial func(ctx context.Context, sess chsession.Session) (*chproto.Codec, error)) *Server {
	return &Server{Hooks: hooks, UpstreamDialer: dial}
}

// Serve accepts loop. Blocks until ln.Accept returns a non-temporary error.
func (s *Server) Serve(ctx context.Context, ln net.Listener) error {
	for {
		c, err := ln.Accept()
		if err != nil {
			return err
		}
		go s.handle(ctx, c)
	}
}

func (s *Server) handle(ctx context.Context, c net.Conn) {
	id := s.nextID.Add(1)
	sess := chsession.New(id, c)
	defer sess.Close()

	up, err := s.UpstreamDialer(ctx, sess)
	if err != nil {
		log.Warnw("dial upstream failed", "id", id, "err", err)
		return
	}
	if err := sess.BindUpstream(ctx, up); err != nil {
		log.Warnw("bind upstream failed", "id", id, "err", err)
		return
	}

	r := NewRelay(sess, s.Hooks)
	if err := r.Run(ctx); err != nil {
		log.Warnw("relay run ended", "id", id, "err", err)
	}
}

// dialUpstreamForCluster returns an UpstreamDialer that borrows a replica
// from the cluster manager for each session. This is the MVP (C1) path:
// one replica per session for the session's lifetime.
func dialUpstreamForCluster(mgr *cluster.Manager) func(context.Context, chsession.Session) (*chproto.Codec, error) {
	return func(ctx context.Context, sess chsession.Session) (*chproto.Codec, error) {
		conn, err := mgr.Borrow(ctx)
		if err != nil {
			return nil, err
		}
		return chproto.NewCodec(conn, chproto.DirToUpstream), nil
	}
}

// dialUpstreamForSidecar / dialUpstreamForForwarding have analogous bodies;
// they wrap the sidecar gRPC or forwarding address chosen at construction.
// Implementations are added in main.go wiring in Step 3.
```
